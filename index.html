async function submitTemporary(){ const list = requireChildren( "temporary" ); if( !list ){ return; } const date = $("temporaryDate").value; const time = $("temporaryTime").value; const reason = $("temporaryReason") .value .trim(); if( !date ){ alert( "請選擇日期。" ); return; } if( isSchoolClosedDate(date) ){ validateSchoolDate( $("temporaryDate") ); return; } if( !reason ){ alert( "請填寫申請原因。" ); return; } /* 測試版暫時依上午班級判斷臨托類別： 櫻桃班、蘋果班 → 幼幼 其他班級 → 混齡 正式版之後應由後端幼兒資料直接判斷， 不應由家長自行選擇。 */ const childTypeLines = list.map( child => { let careType = "混齡"; if( child.classId === "C01" || child.classId === "C02" ){ careType = "幼幼"; } return ( `${child.className} ${child.name}：${careType}` ); } ); const hasToddler = list.some( child => child.classId === "C01" || child.classId === "C02" ); const hasMixed = list.some( child => child.classId !== "C01" && child.classId !== "C02" ); let capacityText = ""; if( hasToddler && hasMixed ){ capacityText = "混齡每日上限8位／幼幼每日上限6位"; } else if( hasToddler ){ capacityText = "幼幼每日上限6位"; } else{ capacityText = "混齡每日上限8位"; } const confirmText = `確認送出臨時托育申請？ 幼兒： ${childrenText(list)} 日期：${date.replaceAll("-","/")} 時段：${time} ${capacityText} ※ 送出後為「待園方確認」， 不代表臨時托育已核准。`; if( !confirm( confirmText ) ){ return; } const msg = `🧸【樂業附幼臨時托育申請】 👶 幼兒： ${childrenText(list)} 📅 日期：${date.replaceAll("-","/")} ⏰ 臨時托育時段：${time} 👧 臨托類別： ${childTypeLines.join("\n")} 📊 本學期臨托名額： 混齡每日最多8位 幼幼每日最多6位 🟣 申請狀態：待園方確認 🏫 課後班級：待園方分配 📝 申請原因： ${reason} 👤 通報家長： ${lineDisplayName || "家長"} ※ 本訊息為臨時托育申請，不代表已核准。 園方確認名額與班級後，將另行通知申請結果。`; try{ await sendLine( msg ); showStatus( "✅ 臨時托育申請已送出。目前狀態為「待園方確認」。園方確認名額後，將再通知核准、候補或無法安排；如核准，也會一併通知課後班級。" ); } catch(e){ console.error( e ); alert( "發送失敗，請確認 LIFF 權限。" ); } }
